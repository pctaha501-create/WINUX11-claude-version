cmake_minimum_required(VERSION 3.24)
project(WINUX11 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
option(WINUX11_BUILD_TESTS "Build WINUX11 tests" ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Qml Quick QuickControls2)
add_subdirectory(compositor)
add_subdirectory(shell)
add_subdirectory(services)
add_subdirectory(compatibility)
add_subdirectory(applications/terminal)
add_subdirectory(applications/explorer)
add_subdirectory(applications/settings)
add_subdirectory(applications/taskmanager)
add_subdirectory(applications/browser)
add_subdirectory(applications/security)
add_subdirectory(applications/screenshot)
add_subdirectory(applications/softwarecenter)
add_subdirectory(applications/network)
add_subdirectory(applications/downloader)
add_subdirectory(applications/archive)
add_subdirectory(applications/calculator)
add_subdirectory(applications/clock)
add_subdirectory(applications/calendar)
add_subdirectory(applications/editor)
add_subdirectory(applications/photos)
add_subdirectory(applications/mediaplayer)
add_subdirectory(applications/music)
add_subdirectory(applications/camera)
add_subdirectory(applications/recorder)
add_subdirectory(applications/systemtools)
if(WINUX11_BUILD_TESTS)
 enable_testing()
 add_subdirectory(tests)
endif()
include(GNUInstallDirs)
install(DIRECTORY session/ DESTINATION ${CMAKE_INSTALL_DATADIR}/wayland-sessions FILES_MATCHING PATTERN "*.desktop")
install(PROGRAMS session/winux11-session DESTINATION ${CMAKE_INSTALL_BINDIR})
