find_package(Qt6WebEngineQuick 6.4 QUIET CONFIG)
if(TARGET Qt6::WebEngineQuick)
  qt_add_executable(winux11-browser main.cpp)
  target_link_libraries(winux11-browser PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::WebEngineQuick)
  qt_add_resources(winux11-browser "browser_qml" FILES Browser.qml)
else()
  message(WARNING "Qt WebEngineQuick is unavailable; building the WINUX11 browser fallback.")
  qt_add_executable(winux11-browser fallback_main.cpp)
  target_link_libraries(winux11-browser PRIVATE Qt6::Core Qt6::Gui)
endif()
install(TARGETS winux11-browser RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
